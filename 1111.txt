// 简单的路由枚举（供预览使用）
enum class Screen { Login, Home, Playlist }

@Composable
fun LinMusicApp() {
    var currentScreen by remember { mutableStateOf(Screen.Login) }
    var isPlayerOpen by remember { mutableStateOf(false) }

    MaterialTheme(colorScheme = darkColorScheme(background = BackgroundDark)) {
        Box(modifier = Modifier.fillMaxSize().background(BackgroundDark)) {

            // 路由切换
            Crossfade(targetState = currentScreen, label = "Screen Transition") { screen ->
                when (screen) {
                    Screen.Login -> LoginScreen(onLogin = { currentScreen = Screen.Home })
                    Screen.Home -> HomeScreen(
                        onNavigateToPlaylist = { currentScreen = Screen.Playlist },
                        onOpenPlayer = { isPlayerOpen = true }
                    )
                    Screen.Playlist -> PlaylistScreen(
                        onBack = { currentScreen = Screen.Home },
                        onOpenPlayer = { isPlayerOpen = true }
                    )
                }
            }

            // 全屏播放器 Modal
            AnimatedVisibility(
                visible = isPlayerOpen,
                enter = slideInVertically(initialOffsetY = { it }, animationSpec = tween(300)),
                exit = slideOutVertically(targetOffsetY = { it }, animationSpec = tween(300)),
                modifier = Modifier.fillMaxSize()
            ) {
                FullPlayerScreen(onClose = { isPlayerOpen = false })
            }
        }
    }
}

@Composable
fun LinBottomNavigation() {
    NavigationBar(containerColor = BackgroundDark, tonalElevation = 0.dp) {
        val items = listOf(
            Triple("Home", Icons.Default.Home, true),
            Triple("Search", Icons.Default.Search, false),
            Triple("Library", Icons.Default.List, false),
            Triple("Profile", Icons.Default.Person, false)
        )
        items.forEach { (label, icon, selected) ->
            NavigationBarItem(
                selected = selected,
                onClick = { },
                icon = { Icon(icon, contentDescription = label) },
                label = { Text(label, fontSize = 10.sp) },
                colors = NavigationBarItemDefaults.colors(
                    selectedIconColor = Color.White, unselectedIconColor = TextGray,
                    selectedTextColor = Color.White, unselectedTextColor = TextGray,
                    indicatorColor = Color.Transparent
                )
            )
        }
    }
}
